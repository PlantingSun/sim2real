cmake_minimum_required(VERSION 3.16)
project(go2w_depth LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CycloneDDS 0.10 REQUIRED)
find_package(realsense2 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(Threads REQUIRED)
idlc_generate(TARGET depth_types FILES DepthFrame.idl DEFAULT_EXTENSIBILITY final)
add_executable(go2w_depth publisher.cpp)
target_include_directories(go2w_depth PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(go2w_depth PRIVATE depth_types CycloneDDS::ddsc realsense2::realsense2 ${OpenCV_LIBS} Threads::Threads)
target_compile_options(go2w_depth PRIVATE -O2 -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(depth_processing_test processing_test.cpp)
target_include_directories(depth_processing_test PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(depth_processing_test PRIVATE realsense2::realsense2 ${OpenCV_LIBS})
add_test(NAME depth_processing COMMAND depth_processing_test)
